# Installing Python versions using Windows Python Install Manager (py or pymanager):
    py list             # Lists Python versions installed and managed by Python Install Manager.
    py list --online    # Lists Python versions available for installation.
    py install 3.12     # Installs Python 3.12 using Python Install Manager.
    py uninstall 3.12   # Uninstalls Python 3.12.

# Creating a Python virtual environment:
    python -m venv venv          # Creates a virtual environment named venv in the current working directory.
    venv\Scripts\activate.bat    # Activates the virtual environment in CMD.
    deactivate                   # Deactivates the active virtual environment.

# Package management within a virtual environment:
    python -m pip list                      # Lists packages installed in the active environment.
    python -m pip install pandas            # Installs pandas in the active environment.
    python -m pip install --upgrade pandas  # Upgrades pandas in the active environment.
    python -m pip uninstall pandas          # Uninstalls pandas from the active environment.
    python -m pip show pandas               # Displays information about an installed package.
    python -m pip freeze                    # Lists installed packages and their versions.

# Common Python Command Line Options
    python -m pip list                          # Execute a module as a script
    python -c "print('Hello')"                  # Execute Python code directly
    python -i script.py                         # Enter interactive mode after executing a script
    python --version                            # Display the Python version
